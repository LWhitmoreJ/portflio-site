A look at the C# scripts behind the gameplay mechanics.

// PlayerMovement.cs
using UnityEngine;
public class PlayerMovement : MonoBehaviour {
    public float moveSpeed = 5f;
    private CharacterController controller;
    void Start() { controller = GetComponent<CharacterController>(); }
    void Update() {
        float h = Input.GetAxisRaw("Horizontal");
        float v = Input.GetAxisRaw("Vertical");
        Vector3 dir = new Vector3(h, 0f, v).normalized;
        controller.Move(dir * moveSpeed * Time.deltaTime);
    }
}
// CombatSystem.cs
using UnityEngine;
public class CombatSystem : MonoBehaviour {
    public int lightDamage = 15;
    public int heavyDamage = 35;
    public float cooldown = 0.4f;
    private float lastAttack;
    void Update() {
        if (Input.GetButtonDown("Fire1") && Time.time - lastAttack >= cooldown)
            Attack(lightDamage, 1.2f);
        if (Input.GetButtonDown("Fire2") && Time.time - lastAttack >= cooldown)
            Attack(heavyDamage, 1.8f);
    }
    void Attack(int dmg, float radius) {
        lastAttack = Time.time;
        Collider[] hits = Physics.OverlapSphere(transform.position + transform.forward, radius);
        foreach (Collider h in hits)
            h.GetComponent<EnemyHealth>()?.TakeDamage(dmg);
    }
}

public class Enemy : MonoBehaviour
{
    .......
    public void Attack()
    {
        Vector3 pos = transform.position;
        pos += transform.right * attackOffset.x;
        pos += transform.up * attackOffset.y;
 
        Collider2D colInfo = Physics2D.OverlapCircle(pos, attackRange, attackMask);
        
        if (colInfo != null)
        {
            colInfo.GetComponent<Player>().TakeDamage(attackDamage);
        }
    }
    public void TakeDamage(int damage)
    {
        //enemy loses health from getting hit with damage
        currentHealth -= damage;
 
        if (!audioSource.isPlaying)
        {
            audioSource.PlayOneShot(hitSound);
        }
        else
        {
            audioSource.Stop();
        }
 
        //Play hurt animation
        animator.SetTrigger("Hurt");
        if (healthBar != null)
        {
            healthBar.SetHealth(currentHealth);
        }
 
        if (currentHealth <= 0)
        {
            Die();
            
            Instantiate(deathEffect, body.position, transform.rotation);
        }
 
        void Die()
        {
            Debug.Log("You have slain your foe!");
            //Play Death animation
 
            animator.SetTrigger("isDead");
            
            //Destroy this game object
            GetComponent<Collider2D>().enabled = false;
            this.enabled = false;
 
            Destroy(this.gameObject, 1f);
        }
    }
}
